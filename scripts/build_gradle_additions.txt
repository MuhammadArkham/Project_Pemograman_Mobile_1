============================================================
  FITME Makan - Tambahkan ke app/build.gradle
============================================================

dependencies {
    // HTTP Client (panggil Anthropic API)
    implementation 'com.squareup.okhttp3:okhttp:4.12.0'

    // Image loading (thumbnail makanan)
    implementation 'com.github.bumptech.glide:glide:4.16.0'
    annotationProcessor 'com.github.bumptech.glide:compiler:4.16.0'

    // Material Design (FAB, dll)
    implementation 'com.google.android.material:material:1.11.0'

    // RecyclerView
    implementation 'androidx.recyclerview:recyclerview:1.3.2'

    // CardView
    implementation 'androidx.cardview:cardview:1.0.0'
}

============================================================
  Tambahkan ke AndroidManifest.xml (di dalam <application>)
============================================================

<uses-permission android:name="android.permission.INTERNET"/>
<uses-permission android:name="android.permission.CAMERA"/>
<uses-permission android:name="android.permission.READ_EXTERNAL_STORAGE"/>

<!-- Di dalam <application> tag: -->

<activity android:name=".com.fitme.app.ui.makan.MakanActivity"/>
<activity android:name=".com.fitme.app.ui.makan.FoodAnalysisActivity"/>

<provider
    android:name="androidx.core.content.FileProvider"
    android:authorities="com.fitme.app.fileprovider"
    android:exported="false"
    android:grantUriPermissions="true">
    <meta-data
        android:name="android.support.FILE_PROVIDER_PATHS"
        android:resource="@xml/file_paths"/>
</provider>
